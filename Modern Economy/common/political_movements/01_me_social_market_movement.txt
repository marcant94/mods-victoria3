movement_social_market = {
	category = movement_category_ideological

	ideology = ideology_social_market_movement
	character_ideologies = {
		ideology_market_liberal
		ideology_reformer
	}

	creation_trigger = {
		OR = {
			has_technology_researched = manufacturies
			has_technology_researched = international_trade
		}
		NOT = {
			AND = {
				has_law_or_variant = law_type:law_social_market_economy
				has_law_or_variant = law_type:law_cooperative_globalism
			}
		}
	}

	creation_weight = {
		value = 100

		if = {
			limit = { literacy_rate > 0.5 }
			multiply = { value = 2 }
		}
	}

	on_created = {
		add_movement_enthusiasm_modifier = yes
	}

	disband_trigger = {
		AND = {
			has_law_or_variant = law_type:law_social_market_economy
			has_law_or_variant = law_type:law_cooperative_globalism
		}
	}

	character_support_trigger = {
		OR = {
			has_ideology = ideology:ideology_market_liberal
			has_ideology = ideology:ideology_reformer
			is_interest_group_type = ig_industrialists
			is_interest_group_type = ig_intelligentsia
		}
	}

	character_support_weight = {
		value = 200
	}

	pop_support_factors = {
		movement_support_high_urbanization
		movement_support_high_literacy
		movement_support_capitalists
		movement_support_shopkeepers
		movement_support_clerks
		movement_support_academics
		movement_support_engineers
	}

	pop_support_weight = {
		add = {
			value = 3
			desc = "POP_BASE_SUPPORT"
		}

		add = {
			desc = "URBAN_STATE"
			value = 10
			multiply = state.state_urbanization_rate
		}

		if = {
			limit = { is_pop_type = capitalists }
			add = {
				value = 40
				desc = "POP_CAPITALISTS"
			}
		}
		else_if = {
			limit = { is_pop_type = shopkeepers }
			add = {
				value = 15
				desc = "POP_SHOPKEEPERS"
			}
		}
		else_if = {
			limit = { is_pop_type = clerks }
			add = {
				value = 10
				desc = "POP_CLERKS"
			}
		}
		else_if = {
			limit = { is_pop_type = academics }
			add = {
				value = 15
				desc = "POP_ACADEMICS"
			}
		}
		else_if = {
			limit = { is_pop_type = engineers }
			add = {
				value = 10
				desc = "POP_ENGINEERS"
			}
		}
	}
}
