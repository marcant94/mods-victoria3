enable_efficient_bureaucracy = {
	is_shown = {
		is_player = yes
		NOT = { has_global_variable = hide_mods_decisions }
		NOT = {
			has_modifier = ebfp_bureaucracy_extra
		}
	}

	possible = {
		is_player = yes
		NOT = {
			has_modifier = ebfp_bureaucracy_extra
		}
	}

	when_taken = {
		add_modifier = {
			name = ebfp_bureaucracy_extra
		}
	}

	ai_chance = {
		value = 0
	}
}

disable_efficient_bureaucracy = {
	is_shown = {
		NOT = { has_global_variable = hide_mods_decisions }
		has_modifier = ebfp_bureaucracy_extra
	}

	possible = {
		has_modifier = ebfp_bureaucracy_extra
	}

	when_taken = {
		remove_modifier = ebfp_bureaucracy_extra
	}

	ai_chance = {
		value = 100
	}
}
