enable_faster_law_enact = {
	is_shown = {
		is_player = yes
		NOT = { has_global_variable = hide_mods_decisions }
		NOT = {
			has_modifier = flfp_faster_law_extra
		}
	}

	possible = {
		is_player = yes
		NOT = {
			has_modifier = flfp_faster_law_extra
		}
	}

	when_taken = {
		add_modifier = {
			name = flfp_faster_law_extra
		}
	}

	ai_chance = {
		value = 0
	}
}

disable_faster_law_enact = {
	is_shown = {
		NOT = { has_global_variable = hide_mods_decisions }
		has_modifier = flfp_faster_law_extra
	}

	possible = {
		has_modifier = flfp_faster_law_extra
	}

	when_taken = {
		remove_modifier = flfp_faster_law_extra
	}

	ai_chance = {
		value = 100
	}
}
